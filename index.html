import React from 'react';

const DevWorkspace = () => {
  const [isLoading, setIsLoading] = React.useState(true);
  
  React.useEffect(() => {
    const timer = setTimeout(() => {
      setIsLoading(false);
    }, 2000);
    
    return () => clearTimeout(timer);
  }, []);
  
  return (
    <div className="workspace">
      {isLoading ? (
        <LoadingScreen />
      ) : (
        <Projects />
      )}
    </div>
  );
};

export default DevWorkspace;
